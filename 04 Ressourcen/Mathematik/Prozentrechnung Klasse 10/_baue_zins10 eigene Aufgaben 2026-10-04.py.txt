NB = " "


def zahl(x, nachkomma=None):
    """Deutsche Schreibweise: Leerzeichen als Tausendertrenner, Komma; ganze Beträge ohne ,00."""
    x = round(x, 2)
    if nachkomma is None:
        nachkomma = 0 if x == int(x) else 2
    t = f"{x:,.{nachkomma}f}".replace(",", " ").replace(".", ",").replace(" ", NB)
    return t


def genau(x, n):
    """Ohne vorheriges Runden auf Cent, n Nachkommastellen, nachgestellte Nullen weg."""
    return f"{x:.{n}f}".rstrip("0").rstrip(".").replace(".", ",")


def eur(x):
    return f"{zahl(x)}{NB}€"


def pz(p):
    """Prozentsatz als Text, z. B. 2,5 %"""
    return f"{zahl(p, 0 if p == int(p) else (1 if round(p * 10) == p * 10 else 2))}{NB}%"


def dez(p):
    """p % als Dezimalzahl, z. B. 0,025"""
    return f"{p / 100:.6f}".rstrip("0").replace(".", ",")

def runde(titel, aufgaben, rezept):
    """Runde: Aufgaben links, das Rezept als Kasten daneben."""

            + f"</ol></div>{rezept}</div>")


def liste(zeilen):
    return '<ol type="a" class="ex">' + "".join(f"<li>{z}</li>" for z in zeilen) + "</ol>"
